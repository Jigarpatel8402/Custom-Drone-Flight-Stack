### setup.py formate:
	<executable_name> = <package_name>.<python_file_name>:main
### launch.py formate:
	Node(
		package = <package_name>,
		executable_name = <executable_name>,
		name = <parameters_name_in_yaml>,
		parameters = [<parameter_file_path>]
	)


### Some Important Commands:

1. Open gazebo world and simulation
gz sim -r $DRONE_PROJECT/simulations/worlds/iris_runway.sdf
 
2. Start ros_gz_bridge
ros2 run ros_gz_bridge parameter_bridge /world/iris_runway/model/iris/link/imu_link/sensor/imu_sensor/imu@sensor_msgs/msg/Imu@gz.msgs.IMU
